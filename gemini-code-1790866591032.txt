    <!-- JavaScript Game Logic -->
    <script>
        // Exact Options Required
        const SECTORS = [
            { label: "5% off", color: "#3B82F6", textColor: "#FFFFFF", icon: "fa-percent" },
            { label: "Vuelve a tirar", color: "#EC4899", textColor: "#FFFFFF", icon: "fa-rotate-right" },
            { label: "10%", color: "#10B981", textColor: "#FFFFFF", icon: "fa-tag" },
            { label: "Game over", color: "#EF4444", textColor: "#FFFFFF", icon: "fa-skull" },
            { label: "Premio", color: "#F59E0B", textColor: "#1E293B", icon: "fa-trophy" },
            { label: "Vuelve a tirar", color: "#8B5CF6", textColor: "#FFFFFF", icon: "fa-rotate-right" },
            { label: "15% off", color: "#06B6D4", textColor: "#FFFFFF", icon: "fa-ticket" },
            { label: "Vuelve a tirar", color: "#6366F1", textColor: "#FFFFFF", icon: "fa-rotate-right" }
        ];

        const numSectors = SECTORS.length;
        const arc = (2 * Math.PI) / numSectors;

        // DOM Elements
        const canvas = document.getElementById("rouletteCanvas");
        const ctx = canvas.getContext("2d");
        const spinBtn = document.getElementById("spinBtn");
        const pointer = document.getElementById("pointer");
        const modal = document.getElementById("resultModal");
        const modalCard = document.getElementById("modalCard");
        const modalTitle = document.getElementById("modalTitle");
        const modalResultText = document.getElementById("modalResultText");
        const modalIconContainer = document.getElementById("modalIconContainer");
        const closeModalBtn = document.getElementById("closeModalBtn");
        const soundToggle = document.getElementById("soundToggle");
        const soundIcon = document.getElementById("soundIcon");
        const soundText = document.getElementById("soundText");
        const spinCountEl = document.getElementById("spinCount");

        // State variables
        let currentAngle = 0; // Angle in radians
        let isSpinning = false;
        let soundEnabled = true;
        let totalSpins = 0;
        let lastPassedSector = -1;

        // Web Audio API Context setup
        let audioCtx = null;

        function initAudio() {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }
        }

        // Sound Generators
        function playTickSound() {
            if (!soundEnabled) return;
            try {
                initAudio();
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                
                osc.type = "sine";
                osc.frequency.setValueAtTime(600, audioCtx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(200, audioCtx.currentTime + 0.04);
                
                gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.04);
                
                osc.connect(gain);
                gain.connect(audioCtx.destination);
                
                osc.start();
                osc.stop(audioCtx.currentTime + 0.04);
            } catch (e) {
                // Audio fallback
            }
        }

        function playFanfareSound(isGameOver) {
            if (!soundEnabled) return;
            try {
                initAudio();
                const now = audioCtx.currentTime;
                
                if (isGameOver) {
                    // Sad/Game Over Tone
                    const osc = audioCtx.createOscillator();
                    const gain = audioCtx.createGain();
                    osc.type = "sawtooth";
                    osc.frequency.setValueAtTime(220, now);
                    osc.frequency.linearRampToValueAtTime(110, now + 0.6);
                    gain.gain.setValueAtTime(0.3, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.6);
                    osc.connect(gain);
                    gain.connect(audioCtx.destination);
                    osc.start(now);
                    osc.stop(now + 0.6);
                } else {
                    // Win Fanfare chord
                    const notes = [440, 554.37, 659.25, 880]; // A major
                    notes.forEach((freq, idx) => {
                        const osc = audioCtx.createOscillator();
                        const gain = audioCtx.createGain();
                        osc.type = "triangle";
                        osc.frequency.setValueAtTime(freq, now + idx * 0.08);
                        gain.gain.setValueAtTime(0.25, now + idx * 0.08);
                        gain.gain.exponentialRampToValueAtTime(0.001, now + idx * 0.08 + 0.5);
                        osc.connect(gain);
                        gain.connect(audioCtx.destination);
                        osc.start(now + idx * 0.08);
                        osc.stop(now + idx * 0.08 + 0.5);
                    });
                }
            } catch (e) {}
        }

        // Setup Canvas Size for High DPI Crispness
        function setupCanvas() {
            const dpr = window.devicePixelRatio || 1;
            const displaySize = Math.min(window.innerWidth - 64, 420);
            
            canvas.width = displaySize * dpr;
            canvas.height = displaySize * dpr;
            canvas.style.width = ${displaySize}px;
            canvas.style.height = ${displaySize}px;
            
            ctx.scale(dpr, dpr);
            drawRoulette();
        }

        // Draw Roulette Canvas
        function drawRoulette() {
            const displayWidth = parseFloat(canvas.style.width);
            const centerX = displayWidth / 2;
            const centerY = displayWidth / 2;
            const radius = displayWidth / 2 - 10;

            ctx.clearRect(0, 0, displayWidth, displayWidth);

            // Draw sectors
            for (let i = 0; i < numSectors; i++) {
                const sectorAngle = currentAngle + i * arc;

                // Slice path
                ctx.beginPath();
                ctx.moveTo(centerX, centerY);
                ctx.arc(centerX, centerY, radius, sectorAngle, sectorAngle + arc);
                ctx.closePath();

                // Fill color
                ctx.fillStyle = SECTORS[i].color;
                ctx.fill();

                // Inner Border Stroke
                ctx.lineWidth = 2;
                ctx.strokeStyle = "rgba(255, 255, 255, 0.25)";
                ctx.stroke();

                // Draw Text & Details
                ctx.save();
                ctx.translate(centerX, centerY);
                ctx.rotate(sectorAngle + arc / 2);

                // Text styling
                ctx.textAlign = "right";
                ctx.fillStyle = SECTORS[i].textColor;
                ctx.font = "bold 16px 'Poppins', sans-serif";
                
                // Shadow for text depth
                ctx.shadowColor = "rgba(0, 0, 0, 0.4)";
                ctx.shadowBlur = 4;
                ctx.shadowOffsetX = 1;
                ctx.shadowOffsetY = 1;

                // Draw label text
                const textDistance = radius - 35;
                ctx.fillText(SECTORS[i].label, textDistance, 6);

                ctx.restore();
            }

            // Draw Center Outer Rim Accent
            ctx.beginPath();
            ctx.arc(centerX, centerY, radius, 0, 2 * Math.PI);
            ctx.lineWidth = 6;
            ctx.strokeStyle = "rgba(255, 255, 255, 0.15)";
            ctx.stroke();
        }

        // Calculate which sector is under the top pointer
        function getSelectedSector() {
            // Pointer is at the top: angle 270 deg (3 * Math.PI / 2)
            const pointerAngle = (3 * Math.PI) / 2;
            let normalizedAngle = (pointerAngle - (currentAngle % (2 * Math.PI)) + 2 * Math.PI) % (2 * Math.PI);
            const index = Math.floor(normalizedAngle / arc);
            return (index % numSectors + numSectors) % numSectors;
        }

        // Spin Functionality with Realistic Acceleration & Ease-Out
        function spin() {
            if (isSpinning) return;
            
            initAudio();
            isSpinning = true;
            spinBtn.disabled = true;
            
            totalSpins++;
            spinCountEl.textContent = totalSpins;

            // Random rotation calculation (minimum 5 full spins + random target angle)
            const minRounds = 5;
            const extraRounds = Math.random() * 3;
            const totalRotation = (minRounds + extraRounds) * 2 * Math.PI + Math.random() * 2 * Math.PI;

            const startTime = performance.now();
            const duration = 4500 + Math.random() * 1000; // ~4.5 - 5.5 seconds
            const startAngle = currentAngle;

            function animateSpin(currentTime) {
                const elapsed = currentTime - startTime;
                const progress = Math.min(elapsed / duration, 1);

                // Cubic Ease-Out for realistic slowdown
                const easeOut = 1 - Math.pow(1 - progress, 3);
                
                currentAngle = startAngle + totalRotation * easeOut;
                drawRoulette();

                // Trigger tick sound & pointer bounce when changing sector
                const currentSector = getSelectedSector();
                if (currentSector !== lastPassedSector) {
                    playTickSound();
                    pointer.classList.remove("pointer-tick");
                    void pointer.offsetWidth; // Trigger reflow for animation restart
                    pointer.classList.add("pointer-tick");
                    lastPassedSector = currentSector;
                }

                if (progress < 1) {
                    requestAnimationFrame(animateSpin);
                } else {
                    isSpinning = false;
                    spinBtn.disabled = false;
                    const winningSector = SECTORS[getSelectedSector()];
                    showResult(winningSector);
                }
            }

            requestAnimationFrame(animateSpin);
        }

        // Show Modal with Results & Visual Effects
        function showResult(winningSector) {
            const isGameOver = winningSector.label === "Game over";
            
            // Set Modal Details
            modalResultText.textContent = winningSector.label;
            
            // Set Icon and Theme based on Result
            if (isGameOver) {
                modalTitle.textContent = "¡Vaya!";
                modalIconContainer.className = "w-20 h-20 mx-auto mb-4 rounded-full flex items-center justify-center text-4xl shadow-lg bg-red-500/20 text-red-500 border border-red-500/40";
                modalIconContainer.innerHTML = <i class="fa-solid ${winningSector.icon}"></i>;
            } else if (winningSector.label.includes("Vuelve a tirar")) {
                modalTitle.textContent = "¡Casi!";
                modalIconContainer.className = "w-20 h-20 mx-auto mb-4 rounded-full flex items-center justify-center text-4xl shadow-lg bg-purple-500/20 text-purple-400 border border-purple-500/40";
                modalIconContainer.innerHTML = <i class="fa-solid ${winningSector.icon}"></i>;
            } else {
                modalTitle.textContent = "¡Felicidades!";
                modalIconContainer.className = "w-20 h-20 mx-auto mb-4 rounded-full flex items-center justify-center text-4xl shadow-lg bg-amber-500/20 text-amber-400 border border-amber-500/40";
                modalIconContainer.innerHTML = <i class="fa-solid ${winningSector.icon}"></i>;
                
                // Launch Confetti for Prizes / Discounts
                if (typeof confetti === 'function') {
                    confetti({
                        particleCount: 80,
                        spread: 70,
                        origin: { y: 0.6 }
                    });
                }
            }

            playFanfareSound(isGameOver);

            // Display Modal
            modal.classList.remove("opacity-0", "pointer-events-none");
            modalCard.classList.remove("scale-90");
            modalCard.classList.add("scale-100");
        }

        function closeModal() {
            modal.classList.add("opacity-0", "pointer-events-none");
            modalCard.classList.remove("scale-100");
            modalCard.classList.add("scale-90");
        }

        // Toggle Sound State
        function toggleSound() {
            soundEnabled = !soundEnabled;
            if (soundEnabled) {
                soundIcon.className = "fa-solid fa-volume-high text-lg";
                soundText.textContent = "Sonido: ON";
            } else {
                soundIcon.className = "fa-solid fa-volume-xmark text-lg text-slate-500";
                soundText.textContent = "Sonido: OFF";
            }
        }

        // Event Listeners
        spinBtn.addEventListener("click", spin);
        closeModalBtn.addEventListener("click", closeModal);
        soundToggle.addEventListener("click", toggleSound);

        // Click outside modal to close
        modal.addEventListener("click", (e) => {
            if (e.target === modal) closeModal();
        });

        // Window Resize Handler
        window.addEventListener("resize", setupCanvas);

        // Initial setup on window load
        window.onload = function() {
            setupCanvas();
        };
    </script>
